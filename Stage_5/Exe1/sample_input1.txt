decl
    int x, y;
    int swap(int *p, int *q);
enddecl

int swap(int *p, int *q)
{
    decl
        int temp;
    enddecl

    begin
        temp = *p;
        *p = *q;
        *q = temp;
        return 0;
    end
}

int main()
{
    decl
        int a, b;
    enddecl

    begin
        a = 10;
        b = 20;
        swap(&a, &b);
        write(a);
        write(b);
        return 0;
    end
}
