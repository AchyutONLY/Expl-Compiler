decl
    int n;
    int a[2][2], b[2][2], c[2][2];
    int i, j, k;
enddecl

read(n);

i = 0;
while (i < n) do
    j = 0;
    while (j < n) do
        read(a[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < n) do
    j = 0;
    while (j < n) do
        read(b[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < n) do
    j = 0;
    while (j < n) do
        c[i][j] = 0;
        k = 0;
        while (k < n) do
            c[i][j] = c[i][j] + a[i][k] * b[k][j];
            k = k + 1;
        endwhile;
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;

i = 0;
while (i < n) do
    j = 0;
    while (j < n) do
        write(c[i][j]);
        j = j + 1;
    endwhile;
    i = i + 1;
endwhile;
